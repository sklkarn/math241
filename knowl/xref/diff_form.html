Standard Differentiation
\begin{align*} \frac{d}{dx}(x^n) \amp = nx^{n-1} \\ \frac{d}{dx}(e^x) \amp = e^x\\ \frac{d}{dx}(a^x) \amp = a^x\ln a\\ \frac{d}{dx}(\ln |x|) \amp = \frac{1}{x}\\ \frac{d}{dx}(\ln |x|) \amp = \frac{1}{x}\\ \frac{d}{dx}(\log_a {x}) \amp = \frac{1}{x\ln a}\\ \frac{d}{dx}(\sin x) \amp = \cos x\\ \frac{d}{dx}(\cos x) \amp = -\sin x\\ \frac{d}{dx}(\tan x) \amp = \sec^2 x\\ \frac{d}{dx}(\csc x) \amp = -\csc x \cot x\\ \frac{d}{dx}(\sec x) \amp = \sec x\tan x\\ \frac{d}{dx}(\cot x) \amp = -\csc^2 x\\ \frac{d}{dx}(\sin^{-1} x) \amp = \frac{1}{\sqrt(1-x^2)}\\ \frac{d}{dx}(\cos^{-1} x) \amp = -\frac{1}{\sqrt(1-x^2)}\\ \frac{d}{dx}(\tan^{-1} x) \amp = \frac{1}{1+x^2}\\ \frac{d}{dx}(\csc^{-1} x) \amp = -\frac{1}{x\sqrt(x^2-1)}\\ \frac{d}{dx}(\sec^{-1} x) \amp = \frac{1}{x\sqrt(x^2-1)}\\ \frac{d}{dx}(\cot^{-1} x) \amp = -\frac{1}{1+x^2} \end{align*}
