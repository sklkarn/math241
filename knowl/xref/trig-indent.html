Trigonometric Identities: in right angled triangle \(\vartriangle ABC,\)
  1. \begin{equation*} h^{2}=p^{2}+b^{2} \end{equation*}
    🔗
  2. \begin{equation*} \sin\theta=\frac{p}{h}; \, \quad \cos\theta=\frac{b}{h}; \, \quad \tan\theta=\frac{p}{b} \end{equation*}
    🔗
  3. \begin{equation*} \sin\theta \csc\theta=1; \, \quad \cos\theta\sec\theta=1; \, \quad \tan\theta\cot\theta=1; \end{equation*}
    🔗
  4. \begin{equation*} \frac{\sin\theta}{\cos\theta}=\tan\theta \end{equation*}
    🔗
    🔗
  5. \begin{equation*} \sin2\theta = 2\sin\theta\cos\theta; \, \quad \cos2\theta = \cos^{2}\theta-\sin^{2}\theta=2\cos^{2}\theta-1=1-2\sin^{2}\theta. \end{equation*}
    \begin{equation*} =2\cos^{2}\theta-1=1-2\sin^{2}\theta. \end{equation*}
    🔗
  6. If \(2\theta = \alpha,\) then
    \begin{equation*} \sin\alpha=2\sin\frac{\alpha}{2}\cdot\cos\frac{\alpha}{2}; \, \quad \cos\alpha=1-2\sin^{2}\frac{\alpha}{2} = 2\cos^{2}\frac{\alpha}{2}-1. \end{equation*}
    🔗
    🔗
  7. \begin{equation*} \sin(-\theta) = -\sin\theta; \, \quad \cos(-\theta) = \cos\theta; \end{equation*}
    🔗
  8. \begin{equation*} \sin(\theta \pm \frac{\pi}{2}) = \pm \cos\theta; \, \quad \cos(\theta \pm \frac{\pi}{2}) = \mp \sin\theta. \end{equation*}
    🔗
    🔗
  9. \begin{equation*} \sin(\alpha\pm\beta)=\sin\alpha\cos\beta\pm\cos\alpha\sin\beta; \end{equation*}
    \begin{equation*} or, \, \cos(\alpha\pm\beta)=\cos\alpha\cos\beta\mp\sin\alpha\sin\beta \end{equation*}
    🔗
  10. \begin{equation*} 2\sin \alpha\cdot\sin\beta = \cos(\alpha-\beta) - \cos(\alpha+\beta); \end{equation*}
    \begin{equation*} or, \, 2\cos \alpha\cdot\cos\beta = \cos(\alpha+\beta) + \cos(\alpha-\beta) \end{equation*}
    🔗
  11. \begin{equation*} 2\sin \alpha\cdot\cos\beta = \sin(\alpha+\beta) + \sin(\alpha-\beta); \end{equation*}
    \begin{equation*} or, \, \quad 2\cos \alpha\cdot\sin\beta = \sin(\alpha+\beta) - \sin(\alpha-\beta) \end{equation*}
    🔗
  12. \begin{equation*} \sin\alpha+\sin\beta = 2\sin\frac{\alpha+\beta}{2}\cos\frac{\alpha-\beta}{2}; \end{equation*}
    \begin{equation*} or, \, \quad \cos\alpha+\cos\beta = 2\cos\frac{\alpha+\beta}{2}\cos\frac{\alpha-\beta}{2} \end{equation*}
    🔗
  13. \begin{equation*} \sin\alpha-\sin\beta = 2\cos\frac{\alpha+\beta}{2}\sin\frac{\alpha-\beta}{2}; \end{equation*}
    \begin{equation*} or, \, \quad \cos\alpha-\cos\beta = -2\sin\frac{\alpha+\beta}{2}\sin\frac{\alpha-\beta}{2} \end{equation*}
    🔗
